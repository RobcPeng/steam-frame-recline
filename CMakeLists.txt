cmake_minimum_required(VERSION 3.16)
project(steam_frame_recline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(src vendor)
find_package(Threads REQUIRED)
add_library(recline_shared SHARED src/driver.cpp)
set_target_properties(recline_shared PROPERTIES PREFIX "driver_" CXX_VISIBILITY_PRESET hidden)
target_link_libraries(recline_shared PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
set(STEAMVR_ROOT /opt/steamvr CACHE PATH "SteamVR installation")
find_library(OPENVR_LIBRARY openvr_api PATHS ${STEAMVR_ROOT}/bin/linuxarm64 NO_DEFAULT_PATH REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf)
add_executable(recline-mode src/main.cpp src/panel.cpp)
target_link_libraries(recline-mode PRIVATE ${OPENVR_LIBRARY} PkgConfig::SDL)
set_target_properties(recline-mode PROPERTIES BUILD_RPATH ${STEAMVR_ROOT}/bin/linuxarm64)
enable_testing()
foreach(name engine lease callback_slot driver)
 add_executable(${name}-test tests/${name}_test.cpp)
 target_compile_options(${name}-test PRIVATE -UNDEBUG)
 target_link_libraries(${name}-test PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
 if(name STREQUAL "driver")
  add_test(NAME driver COMMAND driver-test $<TARGET_FILE:recline_shared>)
  add_test(NAME driver-full COMMAND driver-test $<TARGET_FILE:recline_shared> full)
 else()
  add_test(NAME ${name} COMMAND ${name}-test)
 endif()
endforeach()
